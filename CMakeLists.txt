cmake_minimum_required(VERSION 3.24)
project(n2-cmd VERSION 1.0.0 LANGUAGES C CXX)
if(NOT WIN32)
    message(FATAL_ERROR "n2-cmd requires Windows 10 version 1809 or newer (ConPTY).")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Test)
if(MSVC)
    add_compile_options(/utf-8 /W3)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
endif()

file(GLOB LUA_SOURCES CONFIGURE_DEPENDS third_party/lua/*.c)
add_library(lua54 STATIC ${LUA_SOURCES})
target_include_directories(lua54 PUBLIC third_party/lua)
set_target_properties(lua54 PROPERTIES AUTOMOC OFF)
file(GLOB VTERM_SOURCES CONFIGURE_DEPENDS third_party/libvterm/src/*.c)
add_library(vterm STATIC ${VTERM_SOURCES})
target_include_directories(vterm PUBLIC third_party/libvterm/include PRIVATE third_party/libvterm/src)
set_target_properties(vterm PROPERTIES AUTOMOC OFF)

add_library(n2_state STATIC src/settings_store.cpp src/settings_store.h src/app_types.h)
target_include_directories(n2_state PUBLIC src)
target_link_libraries(n2_state PUBLIC Qt6::Core PRIVATE lua54)
add_library(n2_terminal STATIC src/conpty_session.cpp src/conpty_session.h
    src/terminal_model.cpp src/terminal_model.h src/terminal_view.cpp src/terminal_view.h)
target_link_libraries(n2_terminal PUBLIC n2_state Qt6::Widgets vterm)
add_library(n2_ui STATIC src/main_window.cpp src/main_window.h src/settings_dialog.cpp src/settings_dialog.h
    src/backdrop.cpp src/backdrop.h src/chrome.cpp src/chrome.h)
target_link_libraries(n2_ui PUBLIC n2_terminal)
qt_add_executable(n2-cmd WIN32 src/main.cpp src/resources.qrc src/windows.rc)
target_link_libraries(n2-cmd PRIVATE n2_ui)
target_compile_definitions(n2-cmd PRIVATE N2_CMD_VERSION="${PROJECT_VERSION}")

include(CTest)
if(BUILD_TESTING)
    add_executable(settings_tests tests/settings_tests.cpp src/resources.qrc)
    target_link_libraries(settings_tests PRIVATE n2_state)
    add_test(NAME settings COMMAND settings_tests)
    add_executable(terminal_tests tests/terminal_tests.cpp src/resources.qrc)
    target_link_libraries(terminal_tests PRIVATE n2_terminal Qt6::Test)
    add_test(NAME terminal COMMAND terminal_tests -o "${CMAKE_CURRENT_BINARY_DIR}/terminal-results.txt,txt")
    add_executable(ui_tests tests/ui_tests.cpp src/resources.qrc)
    target_link_libraries(ui_tests PRIVATE n2_ui Qt6::Test)
    add_test(NAME ui COMMAND ui_tests -o "${CMAKE_CURRENT_BINARY_DIR}/ui-results.txt,txt")
    add_executable(conpty_tests tests/conpty_tests.cpp)
    target_link_libraries(conpty_tests PRIVATE n2_terminal Qt6::Test)
    add_test(NAME conpty COMMAND conpty_tests -o "${CMAKE_CURRENT_BINARY_DIR}/conpty-results.txt,txt")
    add_executable(live_tests tests/live_tests.cpp src/resources.qrc)
    target_link_libraries(live_tests PRIVATE n2_ui Qt6::Test)
    add_test(NAME live COMMAND live_tests -o "${CMAKE_CURRENT_BINARY_DIR}/live-results.txt,txt")
    set_tests_properties(settings terminal ui conpty live PROPERTIES TIMEOUT 90)
endif()
install(TARGETS n2-cmd RUNTIME DESTINATION .)
install(DIRECTORY config DESTINATION .)
install(FILES LICENSE README.md DESTINATION .)
install(DIRECTORY licenses DESTINATION .)
